cmake_minimum_required(VERSION 3.16)
project(clipboard-sync VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "clipboard-sync requires Linux and X11/Wayland")
endif()

find_package(Threads REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(XCB REQUIRED IMPORTED_TARGET xcb)
add_library(clipboard_sync_core src/clipboard.cpp src/process.cpp)
target_compile_features(clipboard_sync_core PUBLIC cxx_std_17)
target_include_directories(clipboard_sync_core PUBLIC src PRIVATE third_party/xxhash)
target_link_libraries(clipboard_sync_core PUBLIC Threads::Threads)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  target_compile_definitions(clipboard_sync_core PRIVATE _GNU_SOURCE)
endif()
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(clipboard_sync_core PRIVATE -Wall -Wextra -Wpedantic)
endif()

add_executable(clipboard-sync src/main.cpp)
target_link_libraries(clipboard-sync PRIVATE clipboard_sync_core)
add_executable(clipboard-sync-x11 src/x11_owner.cpp)
target_link_libraries(clipboard-sync-x11 PRIVATE PkgConfig::XCB)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(clipboard-sync-x11 PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_dependencies(clipboard-sync clipboard-sync-x11)

include(GNUInstallDirs)
install(TARGETS clipboard-sync clipboard-sync-x11 RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES clipboard-sync.service DESTINATION lib/systemd/user)
install(FILES LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/clipboard-sync)
install(FILES third_party/xxhash/LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/clipboard-sync RENAME xxhash-LICENSE)
install(FILES README.md DESTINATION ${CMAKE_INSTALL_DOCDIR})

include(CTest)
if(BUILD_TESTING)
  add_executable(clipboard-sync-tests tests/core_test.cpp)
  target_link_libraries(clipboard-sync-tests PRIVATE clipboard_sync_core)
  add_test(NAME core COMMAND clipboard-sync-tests)
  add_executable(clipboard-sync-incr-tests tests/incr_test.cpp)
  target_include_directories(clipboard-sync-incr-tests PRIVATE src)
  add_test(NAME incr COMMAND clipboard-sync-incr-tests)
  find_package(Python3 COMPONENTS Interpreter QUIET)
  if(Python3_Interpreter_FOUND)
    add_test(NAME integration COMMAND ${Python3_EXECUTABLE}
            ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration_test.py $<TARGET_FILE:clipboard-sync>)
    set_tests_properties(integration PROPERTIES TIMEOUT 90)
    add_test(NAME x11_protocol COMMAND ${Python3_EXECUTABLE}
            ${CMAKE_CURRENT_SOURCE_DIR}/tests/x11_protocol_test.py $<TARGET_FILE:clipboard-sync-x11>)
    set_tests_properties(x11_protocol PROPERTIES TIMEOUT 45 SKIP_RETURN_CODE 77)
  endif()
endif()
